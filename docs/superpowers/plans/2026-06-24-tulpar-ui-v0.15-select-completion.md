# Tulpar UI v0.15 — Select completion (`options` data-prop + in-popup search) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish `<tulpar-select>` by adding (1) an `options` data-prop that renders real `<tulpar-option>`/`<tulpar-option-group>` from a typed array, and (2) an opt-in `searchable` in-popup filter with diacritic/Turkish-insensitive matching + match highlighting — both additive, no new public element.

**Architecture:** One small shared-brain change in `_internal/listbox/` (a `hidden` flag through `collection`/`active-index`, plus `searchable`/`hasQuery`/`clear-query` in `keymap`) + a new pure `filter.ts` matcher. The Select renders data-mode options into its own shadow listbox (so the existing styling/active-descendant/click machinery is reused verbatim), generalizes `_options()` to read shadow-or-light, and drives filtering from a single `_filterResult` computed in `willUpdate`. Options/groups gain a `hidden`/highlight hook; `<tulpar-option-group>` gains a `disabled` prop for parity. Angular + Vue wrappers forward the new surface; playgrounds gain parity sections.

**Tech Stack:** Lit 3 + TypeScript (strict, decorators), `@web/test-runner` + `@open-wc/testing` (core, real browser), Vitest + happy-dom (wrapper smoke), Angular v22 signals, Vue 3 SFC, Changesets, pnpm 9 catalogs.

- [ ] **Step 3: Implement `filter.ts`**

```ts
/**
 * Pure substring matcher for a searchable listbox (Select now; Combobox/
 * MultiSelect later). DOM-free + side-effect free so the matching CONTRACT is
 * unit-testable in isolation.
 *
 * The fold is strictly **1:1** (one input char → one output char), NOT a
 * Unicode NFD expansion: output length equals input length, so a match range in
 * normalized space maps back to the ORIGINAL label with an identity index map
 * and highlight spans never shift. (NFD would split `İ` U+0130 into `i`+mark and
 * shift every later index.)
 */

/** Single-codepoint fold table: lowercase + Turkish + common Latin diacritics. */
const FOLD: Record<string, string> = {
  // Turkish
  İ: "i", I: "i", ı: "i", Ş: "s", ş: "s", Ğ: "g", ğ: "g",
  Ç: "c", ç: "c", Ö: "o", ö: "o", Ü: "u", ü: "u",
  // common Latin
  á: "a", à: "a", â: "a", ä: "a", ã: "a", å: "a",
  é: "e", è: "e", ê: "e", ë: "e",
  í: "i", ì: "i", î: "i", ï: "i",
  ó: "o", ò: "o", ô: "o", õ: "o",
  ú: "u", ù: "u", û: "u",
  ñ: "n",
};
// NOTE: every key must be UNIQUE — duplicate object keys are a TS1117 error that
// fails `vite-plugin-dts` at `pnpm build` (ESLint won't catch it). ç/ş/ö/ü/ğ are
// already declared once in the Turkish block above; do NOT repeat them here.

/**
 * Fold a string 1:1. Returns the normalized text plus a map where `map[i]` is the
 * ORIGINAL index of the char that produced normalized char `i` (identity here,
 * kept so the API survives a future non-1:1 fold).
 */
export function normalize(s: string): { text: string; map: number[] } {
  let text = "";
  const map: number[] = [];
  for (let i = 0; i < s.length; i++) {
    const ch = s[i];
    const folded = FOLD[ch] ?? ch.toLowerCase();
    // Guard: keep 1:1 — if a fold ever produces >1 char, take the first (and
    // document that such a fold breaks range alignment, so don't add one).
    text += folded[0] ?? folded;
    map.push(i);
  }
  return { text, map };
}

export interface FilterResult {
  /** Per-label: did the normalized query occur as a substring? */
  matched: boolean[];
  /** Per-label: `[start,end]` in ORIGINAL-label indices, or null (no match / empty query). */
  ranges: Array<[number, number] | null>;
}

/**
 * Match `query` against each label. Empty/whitespace query → all matched, null
 * ranges (no highlight). Otherwise substring (anywhere), diacritic-insensitive.
 */
export function filterOptions(labels: readonly string[], query: string): FilterResult {
  const q = normalize(query.trim()).text;
  if (q === "") {
    return { matched: labels.map(() => true), ranges: labels.map(() => null) };
  }
  const matched: boolean[] = [];
  const ranges: Array<[number, number] | null> = [];
  for (const label of labels) {
    const { text: hay, map } = normalize(label); // normalize ONCE per label
    const at = hay.indexOf(q);
    if (at < 0) {
      matched.push(false);
      ranges.push(null);
    } else {
      matched.push(true);
      const start = map[at];
      const endNorm = at + q.length - 1;
      const end = (map[endNorm] ?? label.length - 1) + 1; // +1 → exclusive end
      ranges.push([start, end]);
    }
  }
  return { matched, ranges };
}
```

```ts
export interface ActiveItem {
  disabled: boolean;
  /** Filtered out (search no-match) — skipped by navigation, like disabled. */
  hidden?: boolean;
}

/** An item is navigable when it is neither disabled nor hidden. */
function navigable(item: ActiveItem | undefined): boolean {
  return !!item && !item.disabled && !item.hidden;
}
```

Then rewrite the bodies to use `navigable`:

```ts
export function firstEnabled(items: readonly ActiveItem[]): number {
  for (let i = 0; i < items.length; i++) if (navigable(items[i])) return i;
  return -1;
}
export function lastEnabled(items: readonly ActiveItem[]): number {
  for (let i = items.length - 1; i >= 0; i--) if (navigable(items[i])) return i;
  return -1;
}
export function nextEnabled(items: readonly ActiveItem[], from: number): number {
  for (let i = from + 1; i < items.length; i++) if (navigable(items[i])) return i;
  if (from >= 0 && from < items.length && navigable(items[from])) return from;
  return from < 0 ? firstEnabled(items) : -1;
}
export function prevEnabled(items: readonly ActiveItem[], from: number): number {
  if (from < 0) return -1;
  for (let i = from - 1; i >= 0; i--) if (navigable(items[i])) return i;
  // navigable(items[from]) is false for an out-of-bounds `from` (the old code used
  // optional chaining and returned `from`); falling back to firstEnabled here is
  // intentionally safer — no existing test exercises a stale out-of-bounds index.
  return navigable(items[from]) ? from : firstEnabled(items);
}
export function pageMove(items: readonly ActiveItem[], from: number, delta: number): number {
  const base = from < 0 ? (delta > 0 ? -1 : items.length) : from;
  let target = base + delta;
  target = Math.max(0, Math.min(items.length - 1, target));
  const dir = delta >= 0 ? 1 : -1;
  for (let i = target; i >= 0 && i < items.length; i += dir) if (navigable(items[i])) return i;
  for (let i = target; i >= 0 && i < items.length; i -= dir) if (navigable(items[i])) return i;
  return from;
}
export function clampActive(items: readonly ActiveItem[], idx: number): number {
  if (idx >= 0 && idx < items.length && navigable(items[idx])) return idx;
  for (let d = 1; d < items.length; d++) {
    const f = idx + d;
    if (f < items.length && navigable(items[f])) return f;
    const b = idx - d;
    if (b >= 0 && b < items.length && navigable(items[b])) return b;
  }
  return firstEnabled(items);
}
```
